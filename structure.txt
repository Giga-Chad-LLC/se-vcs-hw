CheckoutCommand(repository: VCSRepository, revision: String) : VCSCommand
    checkout expressed as:
        1. TreeNode root = HeadManager.loadTreeFromCurrentHead()
        2. List<BLobData> blobs = root.getBlobs();
        3. IndexManager.register(blobs)
        4. IndexManager.updateWorkingDirWithTrackedFiles()

LogCommand(repository: VCSRepository) : VCSCommand
    // sets List<Commit>


MergeCommand(repository: VCSRepository, strategy: IMergeStrategy, sourceBranch: string) : VCSCommand
    merge is expressed as:
        1. List<BLobData> theirBlobs = HeadManager.loadTreeFromCurrentHead(sourceBranch).getBlobs()
        1. List<BLobData> ourBlobs = HeadManager.loadTreeFromCurrentHead().getBlobs()
        1. return strategy.merge(ourBlobs, theirBlobs)

IMergeStrategy:
    merge(ours: List<BLobData>, theirs: List<BLobData>): MergeResult
